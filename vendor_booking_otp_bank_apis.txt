1. VENDOR ACCEPT BOOKING API
================================================================================
Description: Accepts an incoming job booking request assigned to the vendor.

Endpoint: POST /vendors/bookings/action
Full URL: https://tapasyaserver.sreerasthusilvers.co.in/api/vendors/bookings/action
Method: POST
Headers:
  Authorization: Bearer <AUTH_TOKEN>
  Content-Type: application/json

Request Body (JSON):
{
  "booking_id": "BOOKING_ID_HERE",
  "action": "ACCEPTED"
}

Expected Response (200 OK):
{
  "success": true,
  "message": "Booking accepted successfully",
  "data": {
    "id": "BOOKING_ID_HERE",
    "status": "ACCEPTED",
    "updated_at": "2026-08-31T22:00:00Z"
  }
}

Error Response (400 / 404 / 500):
{
  "success": false,
  "message": "Booking already accepted or not found"
}

================================================================================
Description: Sends a 6-digit verification OTP to the customer's mobile number

Endpoint: POST /vendors/bookings/{booking_id}/send-completion-otp
Full URL: https://tapasyaserver.sreerasthusilvers.co.in/api/vendors/bookings/{booking_id}/send-completion-otp
Method: POST
Headers:
  Authorization: Bearer <AUTH_TOKEN>
  Content-Type: application/json

Request Body:
(Empty body - Booking ID is passed in the URL path)

Expected Response (200 OK):
{
  "success": true,
  "message": "OTP sent successfully to customer"
}

Error Response (400 / 500):
{
  "success": false,
  "message": "Failed to send OTP to customer"
}

5. VERIFY JOB COMPLETION OTP API
================================================================================
Description: Verifies the 6-digit OTP provided by customer. Upon verification,
             the booking status becomes COMPLETED and payout is triggered.

Endpoint: POST /vendors/bookings/{booking_id}/verify-otp
Full URL: https://tapasyaserver.sreerasthusilvers.co.in/api/vendors/bookings/{booking_id}/verify-otp
Method: POST
Headers:
  Authorization: Bearer <AUTH_TOKEN>
  Content-Type: application/json

Request Body (JSON):
{
  "booking_id": "BOOKING_ID_HERE",
  "otp": "123456"
}

Expected Response (200 OK):
{
  "success": true,
  "message": "OTP verified successfully. Job completed.",
  "data": {
    "booking_id": "BOOKING_ID_HERE",
    "status": "COMPLETED",
    "payment_status": "PAID"
  }
}

Error Response (400 Invalid OTP):
{
  "success": false,
  "message": "Invalid OTP. Please try again."
}

// Accept Booking
Future<Map<String, dynamic>> acceptBooking(String token, String bookingId, String vendorId) async {
  return bookingAction(token, bookingId, vendorId, "ACCEPTED");
}

// Decline / Reject Booking
Future<Map<String, dynamic>> rejectBooking(String token, String bookingId, String vendorId) async {
  return bookingAction(token, bookingId, vendorId, "REJECTED");
}

// Booking Action Generic Method
Future<Map<String, dynamic>> bookingAction(String token, String bookingId, String vendorId, String action) async {
  final url = Uri.parse('https://tapasyaserver.sreerasthusilvers.co.in/api/vendors/bookings/action');
  final response = await http.post(
    url,
    headers: {
      'Authorization': 'Bearer $token',
      'Content-Type': 'application/json',
    },
    body: jsonEncode({
      "booking_id": bookingId,
      "action": action
    }),
  );
  return jsonDecode(response.body);
}

// Send Completion OTP
Future<Map<String, dynamic>> sendCompletionOtp(String token, String bookingId) async {
  final url = Uri.parse('https://tapasyaserver.sreerasthusilvers.co.in/api/vendors/bookings/$bookingId/send-completion-otp');

    url,

}

// Update & Verify Bank Account
Future<Map<String, dynamic>?> updateBankDetails(
  String token,
  String bankName,
  String accountNumber,
  String ifscCode,
  String holderName,
  String vendorId,
) async {
  final url = Uri.parse('https://tapasyaserver.sreerasthusilvers.co.in/api/vendor/bank/verify');
  final response = await http.post(
    url,
    headers: {
      'Authorization': 'Bearer $token',
      'Content-Type': 'application/json',
    },
    body: jsonEncode({
      'account_number': accountNumber,
      'ifsc': ifscCode,
      'holder_name': holderName,
      'bank_name': bankName,
      'vendor_id': vendorId,
    }),
  );
  return jsonDecode(response.body);
}
